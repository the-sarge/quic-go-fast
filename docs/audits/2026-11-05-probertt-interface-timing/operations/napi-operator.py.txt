"""Owned extension of retained admission/custody operator for issue #1210."""
import hashlib
import importlib.util
import json
from pathlib import Path
import shlex
import shutil
import sys

BASE = Path('/Volumes/worktrees/quic-go-fast/resolve-1210/.cache/1210')

def sha(path):
    return hashlib.sha256(Path(path).read_bytes()).hexdigest()

spec = importlib.util.spec_from_file_location('batch1210', BASE/'batch.py')
batch = importlib.util.module_from_spec(spec)
spec.loader.exec_module(batch)
metadata = BASE/'harness-metadata-witness-linux.test'
name = sys.argv[1]
local_binary, remote_binary, witness = map(Path, sys.argv[2:5])
arms = json.loads((BASE/'arms-napi.json').read_text())
docs = batch.prepare(name, arms, batch.VENUES['eth4'], {'local': local_binary, 'remote': remote_binary})
shutil.copyfile(__file__, docs/'kernel-operator.py.txt')
shutil.copytree(BASE/'napi', docs/'napi')
(docs/'napi-source-hashes.json').write_text(json.dumps({p.name:sha(p) for p in (docs/'napi').iterdir()},indent=2)+'\n')
shutil.copyfile(BASE/'batch.py', docs/'batch.py.txt')
for name in ['bpf-collector.c','bpf-collector','capture-control.py']:
    shutil.copyfile(BASE/name, docs/name)
(docs/'bpf-collector').chmod(0o700)
inputs = {'path':str(witness),'sha256':sha(witness),'test':'(TestDiagnostic1210KernelSocketWitness|TestDiagnostic1210DenseIdentitySegments)','metadata_path':str(metadata),'metadata_sha256':sha(metadata)}
(docs/'kernel-witness.json').write_text(json.dumps(inputs,indent=2)+'\n')
op = batch.load_operator(docs)
batch.install(op, docs)
old_stage, old_capture, old_absence = op.stage, op.capture, op.remote_absence
witness_name = 'kernel-socket-witness.test'
remote_witness = op.REMOTE+'/'+witness_name
remote_metadata = op.REMOTE+'/harness-metadata-witness.test'

def stage(authorization):
    assert sha(witness) == inputs['sha256']
    old_stage(authorization)
    op.run(['scp','-oBatchMode=yes','-oStrictHostKeyChecking=yes','-oConnectTimeout=10',str(witness),'minimax:'+remote_witness], 'stage-kernel-witness', timeout=55)
    source = '''import hashlib,json,pathlib,sys
root=pathlib.Path(sys.argv[1]);name=sys.argv[2];digest=sys.argv[3]
assert root.resolve()==root and root.parent==pathlib.Path('/home/josh') and root.name.startswith('wiremux-1460-diag-q1210-')
p=root/name;assert hashlib.sha256(p.read_bytes()).hexdigest()==digest;p.chmod(0o700)
(root/'kernel-witness').mkdir(mode=0o700)
print(json.dumps({'path':str(p),'sha256':digest}))
'''
    op.remote_python(source,[op.REMOTE,witness_name,inputs['sha256']],'kernel-witness-staged')
    assert sha(metadata) == inputs['metadata_sha256']
    op.run(['scp','-oBatchMode=yes','-oStrictHostKeyChecking=yes','-oConnectTimeout=10',str(metadata),'minimax:'+remote_metadata], 'stage-metadata-witness', timeout=55)
    source = '''import hashlib,json,pathlib,sys
p=pathlib.Path(sys.argv[1]);assert hashlib.sha256(p.read_bytes()).hexdigest()==sys.argv[2];p.chmod(0o700);print(json.dumps({'path':str(p),'sha256':sys.argv[2]}))
'''
    op.remote_python(source,[remote_metadata,inputs['metadata_sha256']],'metadata-witness-staged')

def remote_absence():
    old_absence()
    source = '''import json,pathlib,sys
binary=sys.argv[1];found=[]
for p in pathlib.Path('/proc').iterdir():
 if not p.name.isdigit():continue
 try:
  if str((p/'exe').resolve(strict=True))==binary:found.append(int(p.name))
 except (FileNotFoundError,PermissionError,ProcessLookupError):pass
print(json.dumps({'binary':binary,'live_processes':found}));assert not found
'''
    op.remote_python(source,[remote_witness],'kernel-witness-absence')
    op.remote_python(source,[remote_metadata],'metadata-witness-absence')

def capture():
    command=['env','Q1210_TRACE=1','Q1210_KERNEL_TRACE=1','Q1210_OUTPUT_DIR='+op.REMOTE+'/kernel-witness','timeout','45',remote_witness,'-test.v','-test.run=^'+inputs['test']+'$','-test.count=1']
    op.event('kernel-socket-witness-start',scope='loopback validation; not a native physical-path workload attempt')
    op.run(op.SSH+['minimax',shlex.join(command)],'linux-kernel-socket-witness',timeout=55)
    remote_absence()
    op.event('kernel-socket-witness-passed')
    command=['timeout','45',remote_metadata,'-test.v','-test.run=^TestDiagnostic1210TimestampMetadata$','-test.count=1']
    op.run(op.SSH+['minimax',shlex.join(command)],'linux-metadata-witness',timeout=55)
    remote_absence()
    op.event('kernel-metadata-witness-passed')
    old_capture()

op.stage, op.capture, op.remote_absence = stage, capture, remote_absence
spec_control = importlib.util.spec_from_file_location('capture_control', docs/'capture-control.py')
control = importlib.util.module_from_spec(spec_control)
spec_control.loader.exec_module(control)
active_capture = None
old_local_absence = op.local_absence
spec_napi = importlib.util.spec_from_file_location('napi_adapter', docs/'napi/napi_adapter.py')
napi_module = importlib.util.module_from_spec(spec_napi)
spec_napi.loader.exec_module(napi_module)
napi_captures = {a['name']:napi_module.NapiCapture(op, docs, a['name']) for a in arms if a.get('capture_napi')}
previous_stage = op.stage
previous_remote_absence = op.remote_absence

def stage_with_napi(authorization):
    previous_stage(authorization)
    # Read-only fresh IRQ/module witness before attachment.
    op.run(op.SSH+['minimax', 'cat /sys/class/net/enp97s0/device/irq; cat /proc/irq/66/effective_affinity_list; uname -r; sha256sum /lib/modules/7.0.0-30-generic/kernel/drivers/net/ethernet/realtek/r8169.ko.zst'], 'napi-host-identity', timeout=15)
    for capture in napi_captures.values():
        capture.stage()

def remote_absence_with_napi():
    previous_remote_absence()
    for capture in napi_captures.values():
        capture.remote_absence()

op.stage, op.remote_absence = stage_with_napi, remote_absence_with_napi

def before_arm(arm):
    global active_capture
    if arm.get('capture_bpf'):
        folder = op.ROOT/'bpf'/arm['name']
        folder.parent.mkdir(exist_ok=True)
        active_capture = control.Capture(folder, docs/'bpf-collector')
        op.event('bpf-opening', arm=arm['name'])
        active_capture.start()
        op.event('bpf-ready', arm=arm['name'])
    if arm.get('capture_napi'):
        napi_captures[arm['name']].start()

def after_arm(arm=None):
    global active_capture
    errors = []
    for capture in napi_captures.values():
        if capture.started and not capture.stopped and capture.proc is not None and capture.proc.poll() is None:
            try:
                capture.stop()
            except BaseException as error:
                errors.append(error)
        elif capture.started and not capture.stopped and arm is not None and arm.get('capture_napi'):
            try:
                capture.stop()
            except BaseException as error:
                errors.append(error)
    if active_capture is not None:
        current = active_capture
        try:
            current.stop()
            op.event('bpf-stopped', result=json.loads((current.folder/'collector-result.json').read_text()))
        except BaseException as error:
            errors.append(error)
        finally:
            if current.proc is None or current.proc.poll() is not None:
                active_capture = None
    if errors:
        raise BaseExceptionGroup('capture shutdown errors', errors)

def local_absence():
    after_arm()
    old_local_absence()

op.before_arm, op.after_arm, op.local_absence = before_arm, after_arm, local_absence
op.main()
