package ackhandler

import (
	"context"
	"encoding/json"
	"fmt"
	"os"
	"path/filepath"
	"runtime"
	"runtime/pprof"
	"slices"
	"strconv"
	"testing"
	"time"

	"github.com/quic-go/quic-go/internal/protocol"
)

// TestIssue618Comparison is a frozen driver for one preselected pair cell.
func TestIssue618Comparison(t *testing.T) {
	out := os.Getenv("ISSUE618_COMPARE_OUT")
	if out == "" {
		t.Skip("explicit frozen comparison only")
	}
	if runtime.GOMAXPROCS(0) != 1 {
		t.Fatal("GOMAXPROCS must be 1")
	}
	n, _ := strconv.Atoi(os.Getenv("ISSUE618_N"))
	ranges, _ := strconv.Atoi(os.Getenv("ISSUE618_RANGES"))
	sample, _ := strconv.Atoi(os.Getenv("ISSUE618_SAMPLE"))
	implementation := os.Getenv("ISSUE618_IMPLEMENTATION")
	if (n != 1024 && n != 8192 && n != 32768) || (ranges != 1 && ranges != 32) || sample < 1 || sample > 3 || (implementation != "baseline" && implementation != "candidate") {
		t.Fatal("invalid finite comparison cell")
	}
	started := time.Now()
	f := issue618Build(t, n, ranges)
	for i := 0; i < 16; i++ {
		if i > 0 {
			f.reset()
		}
		ok, err := f.h.ReceivedAck(f.ack, protocol.Encryption1RTT, f.now)
		f.check(t, ok, err)
	}
	var timerP99 int64
	if implementation == "candidate" && n == 1024 && ranges == 1 && sample == 1 {
		var overhead [10000]int64
		for i := range overhead {
			s := time.Now()
			overhead[i] = time.Since(s).Nanoseconds()
		}
		slices.Sort(overhead[:])
		timerP99 = overhead[9899]
	}
	if os.Getenv("ISSUE618_PROFILE") == "1" {
		profile, err := os.Create(filepath.Join(out, "candidate-cpu.pprof"))
		if err != nil {
			t.Fatal(err)
		}
		if err := pprof.StartCPUProfile(profile); err != nil {
			t.Fatal(err)
		}
		ackContext := pprof.WithLabels(context.Background(), pprof.Labels("phase", "ack"))
		resetContext := pprof.WithLabels(context.Background(), pprof.Labels("phase", "reset"))
		end := time.Now().Add(10 * time.Second)
		count := 0
		for time.Now().Before(end) {
			pprof.SetGoroutineLabels(resetContext)
			f.reset()
			pprof.SetGoroutineLabels(ackContext)
			ok, err := f.h.ReceivedAck(f.ack, protocol.Encryption1RTT, f.now)
			pprof.SetGoroutineLabels(resetContext)
			f.check(t, ok, err)
			count++
		}
		pprof.SetGoroutineLabels(context.Background())
		pprof.StopCPUProfile()
		profile.Close()
		t.Logf("profile N=%d ranges=%d calls=%d execution=%s", n, ranges, count, time.Since(started))
		return
	}
	var before, after runtime.MemStats
	var duration time.Duration
	var ops, allocs, bytes uint64
	durations := make([]int64, 0, 3333)
	for duration < 2*time.Second {
		if time.Since(started) > time.Minute {
			t.Fatal("per-sample wall budget reached")
		}
		f.reset()
		runtime.ReadMemStats(&before)
		s := time.Now()
		ok, err := f.h.ReceivedAck(f.ack, protocol.Encryption1RTT, f.now)
		elapsed := time.Since(s)
		runtime.ReadMemStats(&after)
		f.check(t, ok, err)
		duration += elapsed
		ops++
		allocs += after.Mallocs - before.Mallocs
		bytes += after.TotalAlloc - before.TotalAlloc
		if implementation == "candidate" && len(durations) < 3333 {
			durations = append(durations, elapsed.Nanoseconds())
		}
	}
	if implementation == "candidate" {
		tail, err := os.Create(filepath.Join(out, fmt.Sprintf("candidate-tail-%d-%d-%d.csv", n, ranges, sample)))
		if err != nil {
			t.Fatal(err)
		}
		fmt.Fprintln(tail, "sample,operation,elapsed_ns")
		for i, v := range durations {
			fmt.Fprintf(tail, "%d,%d,%d\n", sample, i+1, v)
		}
		tail.Close()
	}
	r := map[string]any{"implementation": implementation, "n": n, "ranges": ranges, "sample": sample, "ops": ops, "ack_ns": duration.Nanoseconds(), "mean_ns": float64(duration.Nanoseconds()) / float64(ops), "allocs": allocs, "bytes": bytes, "tail_count": len(durations), "wall_ns": time.Since(started).Nanoseconds(), "timer_p99_ns": timerP99, "utc": time.Now().UTC()}
	result, err := os.Create(filepath.Join(out, fmt.Sprintf("pair-%d-%d-%d-%s.json", n, ranges, sample, implementation)))
	if err != nil {
		t.Fatal(err)
	}
	if err := json.NewEncoder(result).Encode(r); err != nil {
		t.Fatal(err)
	}
	result.Close()
	t.Logf("%s N=%d ranges=%d sample=%d mean=%dns ops=%d wall=%s", implementation, n, ranges, sample, duration.Nanoseconds()/int64(ops), ops, time.Since(started))
}
