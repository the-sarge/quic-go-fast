// gso742bench is the #742 measurement harness: one process per role, so each
// role's CPU time is its own. It uses only quic-go-fast's public API and is
// built once per source tree (pre-#742, post-#742) with a replace directive.
//
//	receiver: gso742bench -role recv -addr 127.0.0.1:47420 -mode datagram
//	sender:   gso742bench -role send -addr 127.0.0.1:47420 -mode datagram -payload 1190 -duration 10s
//
// Each role prints one JSON line. A verification aid for #742's measurement
// gate, archived with its record; not a maintained deliverable.
package main

import (
	"context"
	"crypto/ecdsa"
	"crypto/elliptic"
	"crypto/rand"
	"crypto/tls"
	"crypto/x509"
	"encoding/binary"
	"encoding/json"
	"encoding/pem"
	"errors"
	"flag"
	"fmt"
	"io"
	"math/big"
	"net"
	"os"
	"syscall"
	"time"

	"github.com/quic-go/quic-go"
)

const alpn = "gso742bench"

func main() {
	role := flag.String("role", "", "send or recv")
	addr := flag.String("addr", "127.0.0.1:47420", "receiver UDP address")
	mode := flag.String("mode", "datagram", "datagram or stream")
	payload := flag.Int("payload", 1190, "DATAGRAM payload bytes")
	duration := flag.Duration("duration", 10*time.Second, "send duration")
	certFile := flag.String("cert", "gso742bench.pem", "receiver certificate, written by recv and trusted by send")
	flag.Parse()
	var out map[string]any
	var err error
	switch *role {
	case "recv":
		out, err = receive(*addr, *mode, *certFile)
	case "send":
		out, err = send(*addr, *mode, *payload, *duration, *certFile)
	default:
		err = fmt.Errorf("unknown role %q", *role)
	}
	if err != nil {
		fmt.Fprintln(os.Stderr, "error:", err)
		os.Exit(1)
	}
	out["role"], out["mode"] = *role, *mode
	out["disable_gso_env"] = os.Getenv("QUIC_GO_DISABLE_GSO")
	line, _ := json.Marshal(out)
	fmt.Println(string(line))
}

func config() *quic.Config {
	return &quic.Config{
		EnableDatagrams:                true,
		InitialStreamReceiveWindow:     4 << 20,
		MaxStreamReceiveWindow:         16 << 20,
		InitialConnectionReceiveWindow: 8 << 20,
		MaxConnectionReceiveWindow:     24 << 20,
		MaxIdleTimeout:                 30 * time.Second,
	}
}

type cpu struct{ user, sys time.Duration }

func now() cpu {
	var ru syscall.Rusage
	_ = syscall.Getrusage(syscall.RUSAGE_SELF, &ru)
	return cpu{time.Duration(ru.Utime.Nano()), time.Duration(ru.Stime.Nano())}
}

func (c cpu) since(start cpu, out map[string]any) {
	out["cpu_user_s"] = (c.user - start.user).Seconds()
	out["cpu_sys_s"] = (c.sys - start.sys).Seconds()
}

func receive(addr, mode, certFile string) (map[string]any, error) {
	udpAddr, err := net.ResolveUDPAddr("udp4", addr)
	if err != nil {
		return nil, err
	}
	udp, err := net.ListenUDP("udp4", udpAddr)
	if err != nil {
		return nil, err
	}
	tr := &quic.Transport{Conn: udp}
	defer tr.Close()
	tlsConf, err := serverTLS(certFile)
	if err != nil {
		return nil, err
	}
	ln, err := tr.Listen(tlsConf, config())
	if err != nil {
		return nil, err
	}
	// The sender waits for this line before dialing.
	fmt.Fprintln(os.Stderr, "ready")
	conn, err := ln.Accept(context.Background())
	if err != nil {
		return nil, err
	}
	out := map[string]any{}
	var bytes, datagrams, maxSeq int64
	var first, last time.Time
	var start cpu
	if mode == "datagram" {
		for {
			b, err := conn.ReceiveDatagram(context.Background())
			if err != nil {
				break
			}
			last = time.Now()
			if datagrams == 0 {
				first, start = last, now()
			}
			datagrams++
			bytes += int64(len(b))
			if len(b) >= 8 {
				maxSeq = max(maxSeq, int64(binary.BigEndian.Uint64(b)))
			}
		}
		out["datagrams"], out["max_seq"] = datagrams, maxSeq
	} else {
		str, err := conn.AcceptUniStream(context.Background())
		if err != nil {
			return nil, err
		}
		buf := make([]byte, 64<<10)
		for {
			n, err := str.Read(buf)
			if n > 0 {
				last = time.Now()
				if bytes == 0 {
					first, start = last, now()
				}
				bytes += int64(n)
			}
			if err == io.EOF {
				break
			}
			if err != nil {
				return nil, err
			}
		}
		conn.CloseWithError(0, "done")
	}
	now().since(start, out)
	elapsed := last.Sub(first)
	out["bytes"], out["elapsed_ns"] = bytes, elapsed.Nanoseconds()
	if elapsed > 0 {
		out["mbps"] = float64(bytes) * 8 / elapsed.Seconds() / 1e6
	}
	return out, nil
}

func send(addr, mode string, payload int, duration time.Duration, certFile string) (map[string]any, error) {
	pemBytes, err := os.ReadFile(certFile)
	if err != nil {
		return nil, err
	}
	roots := x509.NewCertPool()
	if !roots.AppendCertsFromPEM(pemBytes) {
		return nil, errors.New("no certificate in " + certFile)
	}
	udpAddr, err := net.ResolveUDPAddr("udp4", addr)
	if err != nil {
		return nil, err
	}
	udp, err := net.ListenUDP("udp4", &net.UDPAddr{IP: net.IPv4(127, 0, 0, 1)})
	if err != nil {
		return nil, err
	}
	tr := &quic.Transport{Conn: udp}
	defer tr.Close()
	ctx, cancel := context.WithTimeout(context.Background(), 10*time.Second)
	defer cancel()
	conn, err := tr.Dial(ctx, udpAddr, &tls.Config{RootCAs: roots, ServerName: "localhost", NextProtos: []string{alpn}}, config())
	if err != nil {
		return nil, err
	}
	out := map[string]any{}
	start := now()
	begin := time.Now()
	deadline := begin.Add(duration)
	if mode == "datagram" {
		b := make([]byte, payload)
		var seq int64
		for time.Now().Before(deadline) {
			seq++
			binary.BigEndian.PutUint64(b, uint64(seq))
			if err := conn.SendDatagram(b); err != nil {
				return nil, fmt.Errorf("datagram %d: %w", seq, err)
			}
		}
		now().since(start, out)
		out["sent_datagrams"], out["sent_bytes"] = seq, seq*int64(payload)
		out["elapsed_ns"] = time.Since(begin).Nanoseconds()
		// Let the last queued datagrams leave before closing.
		time.Sleep(200 * time.Millisecond)
		conn.CloseWithError(0, "done")
	} else {
		str, err := conn.OpenUniStream()
		if err != nil {
			return nil, err
		}
		b := make([]byte, 64<<10)
		var sent int64
		for time.Now().Before(deadline) {
			n, err := str.Write(b)
			sent += int64(n)
			if err != nil {
				return nil, err
			}
		}
		if err := str.Close(); err != nil {
			return nil, err
		}
		// The receiver closes the connection after reading the whole stream.
		select {
		case <-conn.Context().Done():
		case <-time.After(30 * time.Second):
			return nil, errors.New("receiver did not close")
		}
		now().since(start, out)
		out["sent_bytes"], out["elapsed_ns"] = sent, time.Since(begin).Nanoseconds()
	}
	return out, nil
}

// serverTLS creates an ephemeral self-signed certificate and writes it to
// certFile for the sender to trust.
func serverTLS(certFile string) (*tls.Config, error) {
	key, err := ecdsa.GenerateKey(elliptic.P256(), rand.Reader)
	if err != nil {
		return nil, err
	}
	tmpl := &x509.Certificate{SerialNumber: big.NewInt(1), NotBefore: time.Now().Add(-time.Hour), NotAfter: time.Now().Add(time.Hour), DNSNames: []string{"localhost"}, IsCA: true, BasicConstraintsValid: true, KeyUsage: x509.KeyUsageDigitalSignature | x509.KeyUsageCertSign}
	der, err := x509.CreateCertificate(rand.Reader, tmpl, tmpl, &key.PublicKey, key)
	if err != nil {
		return nil, err
	}
	if err := os.WriteFile(certFile, pem.EncodeToMemory(&pem.Block{Type: "CERTIFICATE", Bytes: der}), 0o600); err != nil {
		return nil, err
	}
	return &tls.Config{Certificates: []tls.Certificate{{Certificate: [][]byte{der}, PrivateKey: key}}, NextProtos: []string{alpn}}, nil
}
