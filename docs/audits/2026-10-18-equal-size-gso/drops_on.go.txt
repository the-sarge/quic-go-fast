//go:build dropcount

package main

import "github.com/quic-go/quic-go"

// receiveDrops reports the receiver tree's drop counters (recvcount.patch).
func receiveDrops() map[string]any {
	datagramQueue, unprocessed := quic.DebugReceiveDropsV743()
	return map[string]any{"datagram_queue_drops": datagramQueue, "unprocessed_drops": unprocessed}
}
