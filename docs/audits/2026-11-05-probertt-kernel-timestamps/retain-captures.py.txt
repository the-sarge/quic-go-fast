from pathlib import Path
import gzip,hashlib,json,shutil,tarfile
ROOT=Path('/Volumes/worktrees/quic-go-fast/kernel-timestamps-1210')
REPORT=ROOT/'docs/audits/2026-11-05-probertt-kernel-timestamps'
CACHE=ROOT/'.cache/1210'

def digest(p):
    h=hashlib.sha256()
    with Path(p).open('rb') as f:
        while b:=f.read(1024*1024):h.update(b)
    return h.hexdigest()

def packed(source,target):
    target.parent.mkdir(parents=True,exist_ok=True)
    with target.open('xb') as f,gzip.GzipFile(filename='',mode='wb',fileobj=f,mtime=0,compresslevel=6) as z:
        shutil.copyfileobj(source,z,1024*1024)

def header(p):
    found=[json.loads(line.split('[DEBUG-1210] ',1)[1]) for line in p.read_text().splitlines() if line.startswith('[DEBUG-1210] {')]
    return found[0] if len(found)==1 else None

    for entry in inv['files']:

    shutil.copytree(inputs,REPORT/'operations'/run,ignore=shutil.ignore_patterns('__pycache__'))
