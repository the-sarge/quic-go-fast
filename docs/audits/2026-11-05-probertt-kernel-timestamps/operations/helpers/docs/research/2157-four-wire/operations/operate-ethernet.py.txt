def sha(path):
    return hashlib.sha256(Path(path).read_bytes()).hexdigest()


def save(name, value, parent=None):
    parent = OPS if parent is None else parent
    path = parent/name
    path.write_text(json.dumps(value, indent=2)+'\n')


def event(kind, **fields):
    with (OPS/'execution-ledger.jsonl').open('a') as output:
        output.write(json.dumps({'kind': kind, 'utc': datetime.datetime.now(datetime.timezone.utc).isoformat(), **fields})+'\n')
        output.flush()
        os.fsync(output.fileno())
    print(kind, json.dumps(fields), flush=True)


def run(argv, label, *, data=None, timeout=30, check=True):
    base = label
    index = 1
    while (OPS/(label+'.command.json')).exists():
        index += 1
        label = base+'-'+str(index)
    started = time.time()
    try:
        result = subprocess.run(argv, input=data, capture_output=True, timeout=timeout)
    except subprocess.TimeoutExpired as error:
        (OPS/(label+'.stdout')).write_bytes(error.stdout or b'')
        (OPS/(label+'.stderr')).write_bytes(error.stderr or b'')
        save(label+'.command.json', {'argv': argv, 'timed_out': True, 'started_unix': started, 'finished_unix': time.time()})
        raise
    (OPS/(label+'.stdout')).write_bytes(result.stdout)
    (OPS/(label+'.stderr')).write_bytes(result.stderr)
    save(label+'.command.json', {'argv': argv, 'exit': result.returncode, 'started_unix': started, 'finished_unix': time.time()})
    if check:
        result.check_returncode()
    return result


def remote_python(source, args, label, host='minimax', check=True):

def acquire_once():
    for host in ADMISSION['hosts_in_acquisition_order']:
        STATE['hosts'][host] = {'status': 'prospective'}
        save('execution-state.json', STATE)
        result = provider(ADMISSION['acquire_action'], host, 'admit-'+host)
        if result.returncode:
            STATE['hosts'][host]['status'] = 'denied' if b'retryable-admission-denial:' in result.stderr else 'uncertain'
            raise RuntimeError('admission '+host+': '+result.stderr.decode(errors='replace'))
        lease = json.loads(result.stdout)
        expected = dict(instance_id=STATE['owner'], host=host, exclusive_host=True, consumer_kind=STATE['consumer_kind'], consumer_id=STATE['consumer_id'])
        assert all(lease.get(key) == value for key, value in expected.items()), lease
        witness = lease_observation(host)['leases'][host]
        assert witness['absent'] is False and witness['lease'] == lease, witness
        STATE['hosts'][host]['status'] = 'held'
        save('lease-'+host+'.json', lease)
        save('execution-state.json', STATE)
        event('admitted', host=host)
    STATE['status'] = 'both-held'
    save('execution-state.json', STATE)


def acquire():
    round_number = 0
    while True:
        observed = preflight_provider()
        if any(observed['leases'].values()):
            event('admission-wait', leases=observed['leases'])
            time.sleep(30)
            continue
        round_number += 1
        STATE['hosts'] = {}
        STATE['status'] = 'acquiring'
        STATE['admission_round'] = round_number
        try:
            acquire_once()
            return
        except BaseException as error:
            save('admission-round-'+str(round_number)+'.json', STATE)
            if 'retryable-admission-denial:' not in str(error):
                raise
            release(True)
            assert STATE['status'] == 'released', STATE
            event('admission-race-unwound', round=round_number)
            time.sleep(30)

def controller_identity(pid):
    return subprocess.check_output(['ps', '-p', str(pid), '-o', 'pid=,lstart='], text=True).strip()


def stop_controller(process, record):
    for sig, grace in ((signal.SIGINT, 5), (signal.SIGTERM, 5), (signal.SIGKILL, 5)):
        if process.poll() is not None:
            return
        assert controller_identity(process.pid) == record['start_identity'], 'controller start identity changed'
        assert os.getpgid(process.pid) == process.pid, 'controller process group changed'
        os.killpg(process.pid, sig)
        record.setdefault('signals', []).append({'signal': sig.name, 'unix': time.time()})
        save('attempt.json', record, ROOT)
        try:
            process.wait(timeout=grace)
            return
        except subprocess.TimeoutExpired:
            pass
    raise RuntimeError('owned controller did not join after bounded escalation')


def capture():
    assert STATE['status'] == 'both-held'
    staged_recheck()
    with (ROOT/'attempt.json').open('x') as output:
        json.dump({'status': 'launch-intent', 'attempt': MANIFEST['proposal_id'], 'config_sha256': sha(DOCS/'capture-config.json')}, output)
    event('attempt-consumed', manifest_sha256=sha(DOCS/'capture-manifest.json'))
    started = time.monotonic()
    record = {'status': 'starting', 'started_unix': time.time(), 'argv': MANIFEST['controller_argv']}
    process = None
    try:
        with (ROOT/'controller.stdout').open('xb') as stdout, (ROOT/'controller.stderr').open('xb') as stderr:
            process = subprocess.Popen(MANIFEST['controller_argv'], stdout=stdout, stderr=stderr, start_new_session=True)
            record['pid'] = process.pid
            record['start_identity'] = controller_identity(process.pid)
            save('attempt.json', record, ROOT)
            record['exit'] = process.wait(timeout=MANIFEST['proposed_limits']['outer_watchdog_seconds'])
            record['status'] = 'returned' if record['exit'] == 0 else 'returned-error'
    except BaseException as error:
        record['status'] = 'interrupted'
        record['error'] = str(error)
        if process is not None and process.poll() is None:
            stop_controller(process, record)
        raise
    finally:
        if process is not None:
            record['exit'] = process.poll()
        record['elapsed_seconds'] = time.monotonic()-started
        record['finished_unix'] = time.time()
        save('attempt.json', record, ROOT)
        event('attempt-ended', **record)


def local_absence():
    binary = MANIFEST['endpoints'][1]['staged_binary']
    library = ctypes.CDLL('/usr/lib/libproc.dylib')
    observed = []
    for line in subprocess.check_output(['ps', '-axo', 'pid='], text=True).splitlines():
        pid = int(line.strip())
        buffer = ctypes.create_string_buffer(4096)
        if library.proc_pidpath(pid, buffer, len(buffer)) > 0 and buffer.value.decode() == binary:
            observed.append({'pid': pid, 'binary': binary, 'start_identity': subprocess.check_output(['ps', '-p', str(pid), '-o', 'lstart='], text=True).strip()})
    save('local-worker-absence.json', {'binary': binary, 'live_processes': observed})
    assert not observed, observed

def verify_archive(archive, inventory):
    expected = {entry['path']: entry for entry in inventory['files']}
    assert len(expected) == len(inventory['files']), 'duplicate inventory entry'
    actual = set()
    with tarfile.open(archive, 'r:gz') as packed:
        for member in packed.getmembers():
            path = Path(member.name)
            assert not path.is_absolute() and '..' not in path.parts
            assert member.isdir() or member.isfile(), member.name
            if member.isfile():
                name = str(path)
                assert name in expected and name not in actual, name
                content = packed.extractfile(member).read()
                assert len(content) == expected[name]['bytes'] and hashlib.sha256(content).hexdigest() == expected[name]['sha256'], name
                actual.add(name)
    assert actual == set(expected)
    return len(actual)

def release(cleanup_ok):
    action = ADMISSION['release_action'] if cleanup_ok else ADMISSION['uncertain_cleanup_action']
    for host, admission in reversed(list(STATE['hosts'].items())):
        if admission['status'] == 'denied':
            continue
        try:
            result = provider(action, host, 'terminal-'+host)
            admission['status'] = 'released' if cleanup_ok and result.returncode == 0 else 'cleanup-retained'
            admission['terminal_exit'] = result.returncode
        except BaseException as error:
            admission['status'] = 'cleanup-retained'
            admission['terminal_error'] = str(error)
        save('execution-state.json', STATE)
    try:
        absence = lease_observation()
        STATE['status'] = 'released' if cleanup_ok and all(entry['absent'] for entry in absence['leases'].values()) and all(entry['status'] in ('released', 'denied') for entry in STATE['hosts'].values()) else 'cleanup-retained'
    except BaseException as error:
        STATE['status'] = 'cleanup-retained'
        STATE['independent_check_error'] = str(error)
        save('execution-state.json', STATE)
        event('cleanup-terminal', status=STATE['status'], error=str(error))
        raise
    save('execution-state.json', STATE)
    event('cleanup-terminal', status=STATE['status'])


def final_inventory():
    files = []
    for path in sorted(ROOT.rglob('*')):
        assert not path.is_symlink(), str(path)
        if path.is_file() and path != ROOT/'final-inventory.json':
            files.append({'path': str(path.relative_to(ROOT)), 'bytes': path.stat().st_size, 'sha256': sha(path)})
    save('final-inventory.json', {'root': str(ROOT), 'files': files, 'excludes': ['final-inventory.json']}, ROOT)


def main():
    authorization = verify_inputs()
    claim_execution(authorization)
    failures = []
    cleanup_ok = False
    try:
        acquire()
        stage(authorization)
        snapshot('before')
        try:
            capture()
        except BaseException as error:
            failures.append(error)
            event('capture-error', error=str(error))
        finally:
            try:
                snapshot('after')
            except BaseException as error:
                failures.append(error)
                event('after-snapshot-error', error=str(error))
    except BaseException as error:
        failures.append(error)
        event('execution-error', error=str(error))
    finally:
        try:
            collect_remove()
            cleanup_ok = True
        except BaseException as error:
            event('cleanup-error', error=str(error))
            failures.append(error)
        try:
            release(cleanup_ok)
        except BaseException as error:
            failures.append(error)
            event('release-error', error=str(error))
        save('execution-errors.json', [{'type': type(error).__name__, 'error': str(error)} for error in failures])
        final_inventory()
    if failures:
        raise BaseExceptionGroup('capture execution and cleanup failures', failures)
    assert STATE['status'] == 'released', STATE


if __name__ == '__main__':
    main()
