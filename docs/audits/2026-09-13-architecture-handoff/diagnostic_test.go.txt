// Diagnostic only: checks the public cancellation promise against current code.
func TestArchitectureProbeCancelAfterPartialReset(t *testing.T) {
 for _, peek := range []bool{false,true} {
  name := "Read"; if peek { name="Peek" }
  t.Run(name,func(t *testing.T){ synctest.Test(t,func(t *testing.T){
   sender:=NewMockStreamSender(gomock.NewController(t))
   sender.EXPECT().onStreamCompleted(protocol.StreamID(42)).AnyTimes()
   s:=newReceiveStream(42,sender,newTestStreamFlowController(42))
   if err:=s.handleResetStreamFrame(&wire.ResetStreamFrame{StreamID:42, ErrorCode:7, FinalSize:10,ReliableSize:6},monotime.Now());err!=nil {t.Fatal(err)}
   done:=make(chan error,1)
   go func(){ var err error; if peek {_,err=s.Peek(make([]byte,1))} else {_,err=s.Read(make([]byte,1))};done<-err }()
   synctest.Wait() // reset notification has been consumed; missing reliable bytes keep reader blocked
   select {case err:=<-done:t.Fatalf("reader returned before local cancellation: %v",err);default:}
   s.CancelRead(9)
   synctest.Wait()
   select {
   case err:=<-done: if err==nil {t.Error("cancelled reader returned nil error")}
   default:
    t.Error("CURRENT CODE VIOLATES CANCELLATION PROMISE: reader remains blocked after CancelRead following partial RESET_STREAM_AT")
    s.closeForShutdown(errors.New("diagnostic cleanup"))
    synctest.Wait()
    <-done
   }
  })})
 }
}

// Diagnostic only: a malformed adapter result must not authorize resubmission.
func TestArchitectureProbeInvalidBatchProgress(t *testing.T) {
 for _, bad:=range []int{-1,3} { t.Run(map[int]string{-1:"negative",3:"over-offered"}[bad],func(t *testing.T){
  c:=&fakeBatchSendConn{accept:func(int,[][]byte)(int,error){return bad,nil}}
  q:=newSendQueue(c,nil).(*sendQueue)
  b1:=getPacketBuffer(); b1.Data=append(b1.Data,1)
  b2:=getPacketBuffer(); b2.Data=append(b2.Data,2)
  err:=q.sendBatchEntries([]queueEntry{{buf:b1},{buf:b2}},c)
  if err==nil {t.Error("CURRENT CODE ACCEPTS MALFORMED PROGRESS: expected fatal contract error")}
  if len(c.writes)!=0 {t.Errorf("CURRENT CODE RETRIES AFTER UNKNOWN PROGRESS: per-packet writes=%d",len(c.writes))}
  if b1.refCount!=0 || b2.refCount!=0 {t.Error("diagnostic buffers not retired")}
 }) }
}
