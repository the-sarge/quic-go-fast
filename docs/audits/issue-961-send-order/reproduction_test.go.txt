package ackhandler

import (
	"testing"
	"time"

	"github.com/quic-go/quic-go/internal/monotime"
	"github.com/quic-go/quic-go/internal/protocol"
	"github.com/quic-go/quic-go/internal/utils"
	"github.com/quic-go/quic-go/internal/wire"
)

// This assessment fixture is activated with go test -overlay, not ordinary CI.
// It asserts correct recovery behavior and is expected to fail on the pinned fork.
func newIssue961Handler() *sentPacketHandler {
	return NewSentPacketHandler(0, 1200, utils.NewRTTStats(), &utils.ConnectionStats{}, true, false, nil, protocol.PerspectiveServer, nil, utils.DefaultLogger).(*sentPacketHandler)
}

func issue961Send(h *sentPacketHandler, now monotime.Time, level protocol.EncryptionLevel) protocol.PacketNumber {
	pn := h.PopPacketNumber(level)
	h.SentPacket(now, pn, protocol.InvalidPacketNumber, nil, []Frame{{Frame: &wire.PingFrame{}}}, level, protocol.ECNNon, 1200, false, false)
	return pn
}

func issue961Ack(t *testing.T, h *sentPacketHandler, now monotime.Time, level protocol.EncryptionLevel, first, last protocol.PacketNumber) {
	t.Helper()
	if _, err := h.ReceivedAck(&wire.AckFrame{AckRanges: []wire.AckRange{{Smallest: first, Largest: last}}}, level, now); err != nil {
		t.Fatal(err)
	}
}

func TestIssue961RecoverySendOrder(t *testing.T) {
	for _, tc := range []struct {
		name  string
		level protocol.EncryptionLevel
	}{
		{"single_space_control", protocol.EncryptionInitial},
		{"overlapping_spaces", protocol.EncryptionHandshake},
	} {
		t.Run(tc.name, func(t *testing.T) {
			t.Run("interleaved_send_order", func(t *testing.T) {
				h := newIssue961Handler()
				now := monotime.Time(time.Second)
				before := h.congestion.GetCongestionWindow()
				var initial, last protocol.PacketNumber
				for i := range 9 {
					initial = issue961Send(h, now, protocol.EncryptionInitial)
					if i%3 == 2 {
						last = issue961Send(h, now, tc.level)
					}
				}
				issue961Ack(t, h, now.Add(20*time.Millisecond), protocol.EncryptionInitial, initial, initial)
				after := h.congestion.GetCongestionWindow()
				t.Logf("interleaved last sent=%s/%d; ACK Initial/%d; lost=%d; cwnd=%d -> %d; recovery=%t", tc.level, last, initial, h.connStats.PacketsLost.Load(), before, after, h.congestion.InRecovery())
				if got := h.connStats.PacketsLost.Load(); got < 4 {
					t.Fatalf("want several threshold losses, got %d", got)
				}
				if want := before * 7 / 10; after != want {
					t.Errorf("interleaved spaces must still cause one cutback per flight: got %d, want %d", after, want)
				}
				if !h.congestion.InRecovery() {
					t.Error("ACK of a pre-cutback interleaved packet must not end recovery")
				}
			})
			t.Run("one_cutback_per_flight", func(t *testing.T) {
				h := newIssue961Handler()
				now := monotime.Time(time.Second)
				before := h.congestion.GetCongestionWindow()
				for range 7 {
					issue961Send(h, now, protocol.EncryptionInitial)
				}
				last := issue961Send(h, now, tc.level)
				issue961Ack(t, h, now.Add(20*time.Millisecond), protocol.EncryptionInitial, 6, 6)
				after := h.congestion.GetCongestionWindow()
				t.Logf("last sent=%s/%d; ACK Initial/6; lost=%d; cwnd=%d -> %d; recovery=%t", tc.level, last, h.connStats.PacketsLost.Load(), before, after, h.congestion.InRecovery())
				if got := h.connStats.PacketsLost.Load(); got != 4 {
					t.Fatalf("want four threshold losses, got %d", got)
				}
				if want := before * 7 / 10; after != want {
					t.Errorf("one recovery flight must cause one Reno cutback: got %d, want %d", after, want)
				}
				if !h.congestion.InRecovery() {
					t.Error("ACK of a pre-cutback packet must not end recovery")
				}
			})
			t.Run("new_flight_loss_cuts_again", func(t *testing.T) {
				h := newIssue961Handler()
				now := monotime.Time(time.Second)
				for range 4 {
					issue961Send(h, now, protocol.EncryptionInitial)
				}
				issue961Ack(t, h, now.Add(20*time.Millisecond), protocol.EncryptionInitial, 1, 3)
				before := h.congestion.GetCongestionWindow()
				var last protocol.PacketNumber
				for range 4 {
					last = issue961Send(h, now.Add(21*time.Millisecond), tc.level)
				}
				issue961Ack(t, h, now.Add(41*time.Millisecond), tc.level, last-2, last)
				after := h.congestion.GetCongestionWindow()
				t.Logf("last sent=%s/%d; lost=%d; second-flight cwnd=%d -> %d", tc.level, last, h.connStats.PacketsLost.Load(), before, after)
				if got := h.connStats.PacketsLost.Load(); got != 2 {
					t.Fatalf("want one loss from each flight, got %d", got)
				}
				if want := before * 7 / 10; after != want {
					t.Errorf("loss of a packet sent after the first cutback must reduce Reno again: got %d, want %d", after, want)
				}
			})
			t.Run("post_cutback_ack_ends_recovery", func(t *testing.T) {
				h := newIssue961Handler()
				now := monotime.Time(time.Second)
				for range 4 {
					issue961Send(h, now, protocol.EncryptionInitial)
				}
				issue961Ack(t, h, now.Add(20*time.Millisecond), protocol.EncryptionInitial, 1, 3)
				if !h.congestion.InRecovery() {
					t.Fatal("setup must enter recovery")
				}
				pn := issue961Send(h, now.Add(21*time.Millisecond), tc.level)
				issue961Ack(t, h, now.Add(41*time.Millisecond), tc.level, pn, pn)
				t.Logf("ACK post-cutback %s/%d; recovery=%t", tc.level, pn, h.congestion.InRecovery())
				if h.congestion.InRecovery() {
					t.Error("ACK of a post-cutback transmission must end recovery")
				}
			})
		})
	}
}
