	"github.com/quic-go/quic-go/internal/ackhandler"

// Throwaway diagnostic for issue #1210; not a qualification or product test.
func TestTriage1210ScheduledProbe(t *testing.T) {
	synctest.Test(t, func(t *testing.T) {
		monotime.Triage1210SetClockOrigin()
		if os.Getenv("TRIAGE1210_ACK") == "1" {
			ackhandler.Triage1210DefaultThreshold = 1
		}
		defer func() { ackhandler.Triage1210DefaultThreshold = 0 }()
		type snapshot struct {
			probe                   bool
			window, flight, pending protocol.ByteCount
			rate                    uint64
			tokens                  int64
			latestRTT, minRTT       time.Duration
			stop                    congestion.SendLimitation
		}
		var state snapshot
		var stateMx sync.Mutex
		triage1210SendObserver = func(c *Conn) {
			if c.perspective != protocol.PerspectiveClient || c.emission.bbr == nil {
				return
			}
			p := c.emission.bbr
			stats := c.ConnectionStats()
			ds := c.sentPacketHandler.(interface {
				DeliveryStats() congestion.DeliveryStats
			}).DeliveryStats()
			_, allowance := c.sentPacketHandler.(interface {
				SendAllowance(monotime.Time) (ackhandler.SendMode, protocol.ByteCount)
			}).SendAllowance(monotime.Now())
			p.credit.mu.Lock()
			pending := p.credit.pending
			p.credit.mu.Unlock()
			s := snapshot{p.controller.InProbeRTT(), p.controller.GetCongestionWindow(), p.controller.GetCongestionWindow() - allowance, pending, p.rate, p.tokens, stats.LatestRTT, stats.MinRTT, ds.Stop}
			stateMx.Lock()
			state = s
			stateMx.Unlock()
		}
		defer func() { triage1210SendObserver = nil }()
		clientPC, serverPC, closeNetwork := newSimnetLink(t, 100*time.Microsecond)
		defer closeNetwork()
		defer clientPC.Close()
		defer serverPC.Close()
		clientTr, serverTr := &Transport{Conn: clientPC}, &Transport{Conn: serverPC}
		defer clientTr.Close()
		defer serverTr.Close()
		serverTLS, clientTLS := silentDropTLSConfigs()
		conf := &Config{EnableDatagrams: true, DisablePathMTUDiscovery: true, InitialPacketSize: 1280}
		require.NoError(t, conf.SetCongestionControlV1("bbrv3"))
		ln, err := serverTr.Listen(serverTLS, conf.Clone())
		require.NoError(t, err)
		defer ln.Close()
		client, err := clientTr.Dial(context.Background(), ln.Addr(), clientTLS, conf.Clone())
		require.NoError(t, err)
		server, err := ln.Accept(context.Background())
		require.NoError(t, err)
		synctest.Wait()
		var received atomic.Uint64
		var admitted, missed uint64
		recvDone := make(chan struct{})
		go func() {
			defer close(recvDone)
			for {
				data, err := server.ReceiveDatagram(context.Background())
				if err != nil {
					return
				}
				received.Add(uint64(len(data)))
			}
		}()
		const payload = 1099
		const rate = 125000000 // application bytes per second: 1 Gbps
		const duration = 5600 * time.Millisecond
		start := time.Now()
		sendDone := make(chan struct{})
		go func() {
			defer close(sendDone)
			data := make([]byte, payload)
			for slot := int64(0); ; slot++ {
				due := start.Add(time.Duration(slot * payload * int64(time.Second) / rate))
				if due.Sub(start) >= duration {
					return
				}
				if delay := time.Until(due); delay > 0 {
					time.Sleep(delay)
				}
				if time.Since(due) > time.Millisecond {
					missed++
					continue
				}
				if err := client.SendDatagram(data); err != nil {
					return
				}
				admitted++
			}
		}()
		type metrics struct {
			n                                                int
			received                                         uint64
			window, flight, pending, rate, tokens, latestRTT float64
			stops                                            map[congestion.SendLimitation]int
		}
		groups := map[bool]*metrics{false: {stops: make(map[congestion.SendLimitation]int)}, true: {stops: make(map[congestion.SendLimitation]int)}}
		var previousReceived uint64
		var wasProbe bool
		for time.Since(start) < duration {
			time.Sleep(100 * time.Microsecond)
			synctest.Wait()
			stateMx.Lock()
			s := state
			stateMx.Unlock()
			probe := s.probe
			receivedNow := received.Load()
			m := groups[probe]
			if time.Since(start) > time.Second {
				m.n++
				m.received += receivedNow - previousReceived
				m.window += float64(s.window)
				m.flight += float64(s.flight)
				m.pending += float64(s.pending)
				m.rate += float64(s.rate)
				m.tokens += float64(s.tokens) / 1e9
				m.latestRTT += float64(s.latestRTT)
				m.stops[s.stop]++
			}
			if probe != wasProbe {
				t.Logf("phase transition t=%s probe=%t window=%d minRTT=%s latestRTT=%s flight=%d pending=%d rate=%d", time.Since(start), probe, s.window, s.minRTT, s.latestRTT, s.flight, s.pending, s.rate)
			}
			wasProbe = probe
			previousReceived = receivedNow
		}
		<-sendDone
		synctest.Wait()
		for _, probe := range []bool{false, true} {
			m := groups[probe]
			n := float64(m.n)
			t.Logf("probe=%t samples=%d deliveredMbps=%.3f avgWindow=%.1f avgFlightCappedAtWindow=%.1f avgPending=%.1f avgPacingMbps=%.3f avgCredit=%.1f avgLatestRTTus=%.3f stops=%v", probe, m.n, float64(m.received)*8/(n*100), m.window/n, m.flight/n, m.pending/n, m.rate/n*8/1e6, m.tokens/n, m.latestRTT/n/1000, m.stops)
		}
		t.Logf("admitted=%d missed=%d received=%d", admitted, missed, received.Load())
		require.Greater(t, groups[true].n, 0, "must exercise ProbeRTT")
		require.NoError(t, client.CloseWithError(0, "diagnostic done"))
		<-recvDone
	})
}
